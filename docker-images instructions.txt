installing postgres with pg vector 

for ubunto
docker run --name postgres \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=postgres123 \
  -p 5432:5432 \
  -v postgres-data:/var/lib/postgresql \
  --restart unless-stopped \
  -d pgvector/pgvector:pg18

docker run --name postgres `
  -e POSTGRES_USER=postgres `
  -e POSTGRES_PASSWORD=postgres123 `
  -p 5432:5432 `
  -v postgres-data:/var/lib/postgresql `
  --restart unless-stopped `
  -d pgvector/pgvector:pg18
